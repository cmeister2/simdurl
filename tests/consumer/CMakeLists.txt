cmake_minimum_required(VERSION 3.20)
option(SIMDURL_CONSUMER_CXX "Build with only a C++ compiler enabled" OFF)
if(SIMDURL_CONSUMER_CXX)
  project(simdurl_consumer LANGUAGES CXX)
else()
  project(simdurl_consumer LANGUAGES C)
endif()

option(SIMDURL_USE_HEADER_ONLY "Consume the inline target instead of the binary" OFF)
find_package(simdurl CONFIG REQUIRED)
if(SIMDURL_USE_HEADER_ONLY)
  set(simdurl_consumer_target simdurl::header_only)
else()
  set(simdurl_consumer_target simdurl::simdurl)
endif()

enable_testing()
add_executable(consumer main.c)
target_link_libraries(consumer PRIVATE ${simdurl_consumer_target})
if(SIMDURL_CONSUMER_CXX)
  set_source_files_properties(main.c PROPERTIES LANGUAGE CXX)
  set_target_properties(consumer PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
  add_executable(consumer_cpp ../cpp_main.cpp ../cpp_other.cpp)
  target_link_libraries(consumer_cpp PRIVATE ${simdurl_consumer_target})
  set_target_properties(consumer_cpp PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
  add_test(NAME installed_cpp_odr COMMAND consumer_cpp)
else()
  set_target_properties(consumer PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
endif()
add_test(NAME installed_consumer COMMAND consumer)

if(WIN32 AND NOT SIMDURL_USE_HEADER_ONLY)
  get_target_property(simdurl_imported_type simdurl::simdurl TYPE)
  if(simdurl_imported_type STREQUAL "SHARED_LIBRARY")
    foreach(target IN ITEMS consumer consumer_cpp)
      if(NOT TARGET ${target})
        continue()
      endif()
      add_custom_command(TARGET ${target} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
          $<TARGET_FILE:simdurl::simdurl> $<TARGET_FILE_DIR:${target}>
        VERBATIM)
    endforeach()
  endif()
endif()
