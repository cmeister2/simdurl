function(simdurl_add_test target library)
  if(ARGC GREATER 2)
    set(test_source ${ARGV2})
  else()
    set(test_source test_simdurl.c)
  endif()
  add_executable(${target} ${test_source})
  target_link_libraries(${target} PRIVATE ${library})
  set_target_properties(${target} PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  if(UNIX)
    target_compile_definitions(${target} PRIVATE SIMDURL_TEST_POSIX=1)
  endif()
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  if(test_source STREQUAL "test_memory.c" AND SIMDURL_TEST_EXTENDED)
    add_test(NAME ${target} COMMAND ${target} --extended)
  else()
    add_test(NAME ${target} COMMAND ${target})
  endif()
  set_tests_properties(${target} PROPERTIES TIMEOUT 180 LABELS "core")
endfunction()
